// End the round in a tie, similar to how an admin can end the round.
VoteType endround
{
   PreflightAction = CallACS("zh2_vote_endround_preflight");
   Action = CallACS("zh2_vote_endround");
   ForbidCVar = sv_zh2_novoteendround;
   DisplayName = "End the round";
   Menu = ZA_CallVote, "End round";
}

// End the round in a tie, similar to how an admin can end the round.
// Additionally, the map will be restarted.
VoteType restartmap
{
   PreflightAction = CallACS("zh2_vote_restartmap_preflight");
   Action = CallACS("zh2_vote_restartmap");
   ForbidCVar = sv_zh2_novoterestartmap;
   DisplayName = "End the round and replay the map";
   Menu = ZA_CallVote, "Restart map";
}

// Allows time to be extended beyond the game's configured time.
VoteType extendtime
{
   PreflightAction = CallACS("zh2_vote_extendtime_preflight");
   Action = CallACS("zh2_vote_extendtime");
   ForbidCVar = sv_zh2_novoteextendtime;
   Arg = int;
   DisplayName = "Extend the round time (seconds)";
   Menu = ZA_CallVote, "Extend round time";
}
